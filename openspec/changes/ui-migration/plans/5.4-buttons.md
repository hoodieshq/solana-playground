## Kind mapping (applies to Tasks 4-7)

| Legacy | Design system |
|---|---|
| `<GradientButton>` | `<BrandButton>` (`variant="ring"`, the default) |
| `kind="primary"` | `<Button>` (`variant="default"`) |
| `kind="secondary"` | `<Button variant="secondary">` |
| `kind="outline"`, `"primary-outline"`, `"secondary-outline"` | `<Button variant="outline">` |
| `kind="transparent"`, `"no-border"`, `"primary-transparent"`, `"secondary-transparent"` | `<Button variant="ghost">` |
| `kind="icon"` | `<Button variant="ghost" size="icon-sm">` (keep `aria-label`/`title`) |
| `kind="error"` | `<Button variant="destructive">` |
| `size="small"` / `"large"` | `size="sm"` / `size="lg"` (`icon-*` for icon buttons) |
| `leftIcon={<X />}` | `<X data-icon="inline-start" />` as the first child |
| `rightIcon={<X />}` | `<X data-icon="inline-end" />` as the last child |
| `fullWidth` | `className="w-full"` |
| `color`, `bg`, `hoverColor`, `fontWeight` | drop, unless the button means something by it (an error colour becomes `destructive`) |
| `onClick` returning a promise, or a `loading` prop | `<AsyncButton onError={reportButtonFailure}>`, `loading={…}`, `loadingText` for `loading.text` |
| `<Button.Copy copyText={v} />` | `<CopyButton value={v} onError={reportButtonFailure} />` |
| `<Button.Export href={h} fileName={f} buttonKind={k}>T</Button.Export>` | `<Button asChild variant={…}><a href={PgCommon.getDataUrl(h)} download={f}>T</a></Button>` |
| `<Button.Import …>` | `<ImportButton …>` from `components/ImportButton` (Task 3) |
| `styled(Button)` / `styled(GradientButton)` | the DS component plus `className`; a styled wrapper that only sets layout stays on the wrapper element, not the button |

How to tell a promise-returning handler: the handler is `async`, returns `PgCommand.*.execute()`, or returns another call typed `Promise`. When in doubt, read the callee's return type; the type checker does not tell the two apart.

## Review Focus

1. A handler that throws synchronously inside `AsyncButton`: the button must stay enabled and the error reach `onError` (pin in Task 1).
2. An `AsyncButton` that unmounts before its promise settles (a dialog's submit closes the dialog): no React warning, no state set after unmount (pin in Task 1).
3. `CopyButton` pressed twice inside the 1.4 s window: "Copied" stays, no stale timer flips it back early (pin in Task 1).
4. Icon buttons losing their accessible name in the move (`kind="icon"` sites that relied on `title`): every icon button keeps an `aria-label` or `title` (pin in Task 8's grep and the existing e2e).
5. A `styled(Button)` rule that out-ranks the DS classes (styled CSS is unlayered and wins): after Task 4-7, no `styled(Button…)` remains (Task 8 grep).

`design-system/src/components/ui/async-button.tsx`, replace the component body:

```tsx
/* A Button that waits for its own work. Give it an onClick that returns a
   promise and it disables itself and shows a spinner until the promise
   settles, so a slow build cannot be started twice. loading takes over when
   the wait is known elsewhere. A rejection or a throw goes to onError; with
   no onError it is rethrown, so it is never lost. It stays a Button: same
   variants, same slot. */
function AsyncButton({
  onClick,
  onError,
  loading: loadingProp,
  loadingText,
  disabled,
  children,
  ...props
}: Omit<React.ComponentProps<typeof Button>, "onClick" | "onError"> & {
  onClick?: (event: React.MouseEvent<HTMLButtonElement>) => unknown
  /** where a failed click goes; without it the failure is rethrown */
  onError?: (error: unknown) => void
  loading?: boolean
  /** what the button says while it waits, e.g. "Building" */
  loadingText?: React.ReactNode
}) {
  const [pending, setPending] = React.useState(false)
  const mounted = React.useRef(true)
  React.useEffect(() => {
    mounted.current = true
    return () => {
      mounted.current = false
    }
  }, [])
  const loading = loadingProp ?? pending
  return (
    <Button
      data-async=""
      data-loading={loading || undefined}
      aria-busy={loading || undefined}
      disabled={disabled || loading}
      onClick={async (event) => {
        try {
          const result = onClick?.(event)
          if (result instanceof Promise) {
            setPending(true)
            await result
          }
        } catch (error) {
          if (!onError) throw error
          onError(error)
        } finally {
          // A submit that closes its dialog unmounts the button first
          if (mounted.current) setPending(false)
        }
      }}
      {...props}
    >
      {loading && <Spinner data-icon="inline-start" />}
      {loading && loadingText ? loadingText : children}
    </Button>
  )
}
```

`design-system/src/components/ui/copy-button.tsx`, replace the component:

```tsx
/* Copies one value: a program id, a signature, a command. Its tooltip turns
   from Copy to Copied, in green, and back. A clipboard that refuses (or a
   browser with none) goes to onError, rethrown without one, and never
   shows Copied. */
function CopyButton({
  value,
  label = "Copy",
  onError,
  className,
  ...props
}: Omit<React.ComponentProps<typeof Button>, "value" | "onClick" | "onError"> & {
  value: string
  label?: string
  /** where a refused copy goes */
  onError?: (error: unknown) => void
}) {
  const [open, setOpen] = React.useState(false)
  const [copied, setCopied] = React.useState(false)
  const timer = React.useRef<ReturnType<typeof setTimeout>>(undefined)
  React.useEffect(() => () => clearTimeout(timer.current), [])
  const copy = async () => {
    try {
      if (!navigator.clipboard) throw new Error("No clipboard in this browser")
      await navigator.clipboard.writeText(value)
    } catch (error) {
      if (!onError) throw error
      onError(error)
      return
    }
    setCopied(true)
    clearTimeout(timer.current)
    timer.current = setTimeout(() => setCopied(false), 1400)
  }
  return (
    <Tooltip open={open || copied} onOpenChange={setOpen}>
      <TooltipTrigger asChild>
        <Button
          data-slot="copy-button"
          variant="ghost"
          size="icon-xs"
          aria-label={copied ? "Copied" : label}
          className={cn("text-muted-foreground hover:text-foreground", className)}
          onClick={copy}
          {...props}
        >
          {copied ? <Check className="text-success" /> : <Copy />}
        </Button>
      </TooltipTrigger>
      <TooltipContent>
        <span className={copied ? "text-success" : undefined}>{copied ? "Copied" : label}</span>
      </TooltipContent>
    </Tooltip>
  )
}
```
